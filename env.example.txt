# ---------------------------------------------------------------------------
# fast-remote-gpu configuration
# Copy to .env:   cp env.example.txt .env   then fill in your keys.
# (kept as .txt so it is never treated as a live secrets file)
# ---------------------------------------------------------------------------

# Provider: "vast" (primary / cheapest) or "runpod" (reliable fallback)
PROVIDER=vast
VAST_API_KEY=
RUNPOD_API_KEY=

# Image to run on the remote box (built & pushed by GitHub Actions to GHCR)
# Replace your-github-username with yours.
IMAGE=ghcr.io/your-github-username/fast-remote-gpu:latest

# If the image is PRIVATE, set GHCR_USER + a token with read:packages.
GHCR_USER=
GHCR_TOKEN=

# Mode: "interruptible" (cheap, preemptible) or "on_demand" (stable)
MODE=on_demand

# GPU allow-list (comma separated). Cheapest available match wins.
GPU_NAMES=RTX 5060 Ti,RTX 5070,RTX 5070 Ti,RTX 5080,RTX 5090,RTX 4090,RTX 4080 SUPER,RTX 4080,RTX 3090,RTX 3080

# Max price per hour (USD).
MAX_PRICE=0.60

# Persistent workspace volume size (GB)
DISK_GB=100

# SSH key used to reach the box. Auto-generated on first up.sh if missing.
SSH_KEY=~/.ssh/fast_remote_gpu

# Vast account SSH key handling:
#   auto  = register SSH_KEY on your Vast account automatically each run (default)
#   skip  = do not touch Vast keys (you added it manually at vast.ai → Account → SSH Keys)
#   /path/to/key = register THIS key (public part) instead of SSH_KEY
VAST_SSH_KEY=auto

# Local file storing the current instance id
STATE_FILE=.instance

# Ports forwarded locally by ssh.sh
TENSORBOARD_PORT=6006
JUPYTER_PORT=8888

# Virtualenv on the remote /workspace:
#   0 = use the image's SYSTEM python (default; shortest boot, driver-matched torch)
#   1 = also create /workspace/.venv. It REUSES the image's torch/torchvision
#       (never reinstalls them if already present), installing only the rest.
USE_VENV=0

# Auto-open the remote /workspace in a new VS Code window when up.sh finishes.
#   1 = open automatically (needs the 'code' CLI on PATH)  0 = just print the command
OPEN_VSCODE=1
